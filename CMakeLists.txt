cmake_minimum_required(VERSION 3.24)
project(NovaEditor VERSION 0.1.0 LANGUAGES CXX)
set(CMAKE_CXX_STANDARD 17)
set(CMAKE_CXX_STANDARD_REQUIRED ON)
# Migrate only the old default; retain explicit external SDK roots.
set(_nova_legacy_sdk "${CMAKE_CURRENT_SOURCE_DIR}/.sdk/native")
if(DEFINED NOVA_SDK_ROOT)
  cmake_path(COMPARE "${NOVA_SDK_ROOT}" EQUAL "${_nova_legacy_sdk}" _nova_old_default)
  if(_nova_old_default)
    set(NOVA_SDK_ROOT "${CMAKE_CURRENT_SOURCE_DIR}/deps/nova/native" CACHE PATH "Installed Nova SDK prefix" FORCE)
  endif()
endif()
get_cmake_property(_nova_cached CACHE_VARIABLES)
foreach(_nova_var IN LISTS _nova_cached)
  if(_nova_var MATCHES "_DIR$")
    cmake_path(IS_PREFIX _nova_legacy_sdk "${${_nova_var}}" NORMALIZE _nova_old_import)
    if(_nova_old_import)
      unset(${_nova_var} CACHE)
    endif()
  endif()
endforeach()
set(NOVA_SDK_ROOT "${CMAKE_CURRENT_SOURCE_DIR}/deps/nova/native" CACHE PATH "Installed Nova SDK prefix")
include("${CMAKE_CURRENT_SOURCE_DIR}/tools/sdk/check-pair.cmake")
list(PREPEND CMAKE_PREFIX_PATH "${NOVA_SDK_ROOT}")
find_package(Nova 0.5 CONFIG REQUIRED)
find_package(imgui CONFIG REQUIRED)
set(JSON_INCLUDE_DIR "${CMAKE_CURRENT_SOURCE_DIR}/third_party/nlohmann-json/include")
add_library(editor_core INTERFACE)
target_include_directories(editor_core INTERFACE apps/desktop/src "${JSON_INCLUDE_DIR}")
add_executable(nova_editor apps/desktop/src/main.cpp)
target_link_libraries(nova_editor PRIVATE editor_core Nova::Render imgui::imgui glfw)
if(APPLE)
  set_target_properties(nova_editor PROPERTIES MACOSX_BUNDLE TRUE MACOSX_BUNDLE_GUI_IDENTIFIER "org.nova.editor" MACOSX_BUNDLE_BUNDLE_NAME "Nova Editor")
endif()
