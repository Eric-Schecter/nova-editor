cmake_minimum_required(VERSION 3.24)
if(CMAKE_SOURCE_DIR STREQUAL CMAKE_CURRENT_SOURCE_DIR)
  project(NovaFBX LANGUAGES C CXX)
endif()
include(FetchContent)
set(BUILD_SHARED_LIBS OFF)
set(ASSIMP_BUILD_TESTS OFF)
set(ASSIMP_BUILD_ASSIMP_TOOLS OFF)
set(ASSIMP_INSTALL OFF)
set(ASSIMP_WARNINGS_AS_ERRORS OFF)
set(ASSIMP_BUILD_ALL_IMPORTERS_BY_DEFAULT OFF)
set(ASSIMP_BUILD_ALL_EXPORTERS_BY_DEFAULT OFF)
set(ASSIMP_BUILD_FBX_IMPORTER ON)
set(ASSIMP_BUILD_FBX_EXPORTER ON)
set(ASSIMP_BUILD_GLTF_IMPORTER ON)
set(ASSIMP_BUILD_GLTF_EXPORTER ON)
if(EMSCRIPTEN)
  set(ASSIMP_BUILD_ZLIB ON)
  set(Z_HAVE_UNISTD_H ON CACHE INTERNAL "Emscripten libc provides unistd.h" FORCE)
  add_compile_options(-fexceptions)
endif()
FetchContent_Declare(assimp GIT_REPOSITORY https://github.com/assimp/assimp.git GIT_TAG v6.0.4 GIT_SHALLOW TRUE)
FetchContent_MakeAvailable(assimp)
if(NOT CMAKE_SOURCE_DIR STREQUAL CMAKE_CURRENT_SOURCE_DIR)
  set(assimp_SOURCE_DIR "${assimp_SOURCE_DIR}" PARENT_SCOPE)
endif()
add_executable(nova_fbx convert.cpp)
target_compile_features(nova_fbx PRIVATE cxx_std_17)
target_include_directories(nova_fbx PRIVATE ../../apps/desktop/src ../../third_party/nlohmann-json/include)
target_link_libraries(nova_fbx PRIVATE assimp::assimp)
if(EMSCRIPTEN)
  set_target_properties(nova_fbx PROPERTIES SUFFIX ".mjs")
  target_link_options(nova_fbx PRIVATE -sMODULARIZE=1 -sEXPORT_ES6=1
    -sINVOKE_RUN=0 -sFORCE_FILESYSTEM=1 -sALLOW_MEMORY_GROWTH=1
    "-sEXPORTED_RUNTIME_METHODS=['FS','callMain']")
  target_link_options(nova_fbx PRIVATE -fexceptions)
endif()
